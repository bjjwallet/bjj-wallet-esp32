cmake_minimum_required(VERSION 3.20)
project(airlock_portable LANGUAGES C CXX)

# Host-side checks for the portable part of the ESP32 migration.
# This target does not produce an ESP32 firmware image.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
file(GLOB AIRLOCK_LANGUAGE_CATALOGS "${CMAKE_CURRENT_SOURCE_DIR}/ui/locales/*.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${AIRLOCK_LANGUAGE_CATALOGS}
    "${CMAKE_CURRENT_SOURCE_DIR}/ui/translations.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_ui_strings.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../ui/states_menu.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../ui/states_wallet.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../ui/states_ble.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../ui/state_sign.py")
execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_words.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/../words.txt" "${GENERATED_DIR}/wordlist.hpp"
    COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_ui_strings.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/.." "${GENERATED_DIR}/original_strings.hpp" COMMAND_ERROR_IS_FATAL ANY)
add_library(airlock_json core/json_memory.cpp vendor/cjson/cJSON.c)
target_include_directories(airlock_json PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cjson")
add_library(airlock_core
    core/rules.cpp
    core/public_wallet_rpc.cpp
    hardware/ble_framing.cpp
    hardware/ft6336.cpp
    ui/states_wallet.cpp
    ui/state_sign.cpp)
target_include_directories(airlock_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${GENERATED_DIR}")
target_link_libraries(airlock_core PUBLIC airlock_json)
# std::function crosses the framing/RPC boundary. Its inline definitions must
# use the same RTTI configuration in the core, adapters and test executables.
target_compile_options(airlock_core PUBLIC -fno-rtti PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)

option(AIRLOCK_SANITIZERS "Run host checks with address/undefined behavior sanitizers" OFF)
if(AIRLOCK_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(airlock_json PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(airlock_json PUBLIC -fsanitize=address,undefined)
    target_compile_options(airlock_core PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(airlock_core PUBLIC -fsanitize=address,undefined)
endif()

# Optional real LVGL renderer checks; no dependency downloads at configure time.
set(AIRLOCK_LVGL_DIR "" CACHE PATH "Local LVGL 9.2.2 source checkout")
option(AIRLOCK_SIMULATOR "Build the interactive SDL2 interface preview" OFF)
if(AIRLOCK_SIMULATOR AND NOT AIRLOCK_LVGL_DIR)
    message(FATAL_ERROR "The simulator needs AIRLOCK_LVGL_DIR pointing to LVGL 9.2.2")
endif()
if(AIRLOCK_LVGL_DIR)
    file(READ "${AIRLOCK_LVGL_DIR}/lv_version.h" AIRLOCK_LVGL_VERSION)
    if(NOT AIRLOCK_LVGL_VERSION MATCHES "LVGL_VERSION_MAJOR +9" OR
       NOT AIRLOCK_LVGL_VERSION MATCHES "LVGL_VERSION_MINOR +2" OR
       NOT AIRLOCK_LVGL_VERSION MATCHES "LVGL_VERSION_PATCH +2")
        message(FATAL_ERROR "The renderer is pinned to LVGL 9.2.2; pass that source directory.")
    endif()
    enable_language(C)
    set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/ui/lv_conf.h" CACHE STRING "" FORCE)
    set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
    set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
    set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
    add_subdirectory("${AIRLOCK_LVGL_DIR}" lvgl)
    if(AIRLOCK_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(lvgl PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    endif()
    add_library(airlock_touch ui/touch_view.cpp ui/lvgl_memory.cpp ui/wallet_app.cpp ui/first_login.cpp ui/assets/splash_image.c ui/fonts/airlock_chinese_16.c)
    target_include_directories(airlock_touch PRIVATE "${GENERATED_DIR}")
    target_link_libraries(airlock_touch PUBLIC airlock_core lvgl::lvgl)
    target_compile_options(airlock_touch PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(BUILD_TESTING)
        add_executable(airlock_touch_tests tests/touch_tests.cpp)
        target_link_libraries(airlock_touch_tests PRIVATE airlock_touch)
        add_test(NAME touch_view COMMAND airlock_touch_tests)
        add_executable(airlock_board_serial_tests tests/board_serial_tests.cpp ui/board_serial_control.cpp)
        target_link_libraries(airlock_board_serial_tests PRIVATE airlock_touch)
        target_compile_options(airlock_board_serial_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME board_serial_control COMMAND airlock_board_serial_tests)
        add_test(NAME board_agent_protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/board_agent_tests.py")
        if(TARGET airlock_crypto)
            add_executable(airlock_first_login_tests tests/first_login_tests.cpp ui/board_serial_control.cpp)
            target_link_libraries(airlock_first_login_tests PRIVATE airlock_touch airlock_crypto)
            target_compile_options(airlock_first_login_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)

            set_tests_properties(hardware_access_flow PROPERTIES ENVIRONMENT "AIRLOCK_TEST_ACCESS_ONLY=1")
        endif()
    endif()
    if(AIRLOCK_SIMULATOR)
        find_package(SDL2 REQUIRED)
        find_package(OpenSSL REQUIRED)
        find_package(SQLite3 REQUIRED)
        find_package(Threads REQUIRED)
        set(AIRLOCK_WALLET_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/wallet-core-4.4.2" CACHE PATH "Official Wallet Core 4.4.2 Apple distribution")

        if(NOT APPLE OR NOT EXISTS "${AIRLOCK_WALLET_CORE_LIB}")

        target_include_directories(airlock_native_wallet SYSTEM PRIVATE "${AIRLOCK_WALLET_CORE_DIR}/include" "${AIRLOCK_WALLET_CORE_DIR}/trezor-include")

        target_compile_options(airlock_native_wallet PRIVATE -Wall -Wextra -Werror)
        set_source_files_properties(vendor/cjson/cJSON.c PROPERTIES COMPILE_OPTIONS "-Wno-error=deprecated-declarations")
